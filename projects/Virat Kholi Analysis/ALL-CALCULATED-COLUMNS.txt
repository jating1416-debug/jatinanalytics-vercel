Dismissal Show =
IF ( 'fact_innings'[dismissal] = "", "—", 'fact_innings'[dismissal] )

score bin =
SWITCH ( TRUE (),
    'fact_innings'[runs] >= 100, "100+",
    'fact_innings'[runs] >= 50,  "50-99",
    'fact_innings'[runs] >= 30,  "30-49",
    'fact_innings'[runs] >= 10,  "10-29",
    "0-9"
)
// agar tere bin boundaries thode alag hain to apna wala hi rakhna

Bin Order =
CALCULATE (
    MIN ( 'fact_innings'[runs] ),
    ALLEXCEPT ( 'fact_innings', 'fact_innings'[score bin] )
)
// phir score bin column → Column tools → Sort by column → Bin Order
